1. Wait for it to come up:

   kubectl rollout status deploy/{{ include "fastapiredis.fullname" . }}-app -n {{ .Release.Namespace }}

2. Reach the app:
{{- if eq .Values.app.service.type "NodePort" }}

   ⚠️  This is a kind-backed cluster: NodePort is NOT reachable from macOS
       (see step 6). Either port-forward, or run the socat bridge:

   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "fastapiredis.fullname" . }}-app 8080:{{ .Values.app.service.port }}
   # -> http://localhost:8080

   # or, for a stable NodePort URL:
   NODE_IP=$(kubectl get nodes -o jsonpath='{.items[1].status.addresses[0].address}')
   docker run -d --name k8s-np-proxy --network kind -p {{ .Values.app.service.nodePort }}:{{ .Values.app.service.nodePort }} \
     alpine/socat tcp-listen:{{ .Values.app.service.nodePort }},fork,reuseaddr tcp-connect:$NODE_IP:{{ .Values.app.service.nodePort }}
   # -> http://localhost:{{ .Values.app.service.nodePort }}
{{- else if eq .Values.app.service.type "LoadBalancer" }}

   kubectl get svc -n {{ .Release.Namespace }} {{ include "fastapiredis.fullname" . }}-app -w
   # wait for EXTERNAL-IP, then http://<EXTERNAL-IP>:{{ .Values.app.service.port }}
{{- else }}

   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "fastapiredis.fullname" . }}-app 8080:{{ .Values.app.service.port }}
   # -> http://localhost:8080
{{- end }}

3. Try it:

   curl localhost:8080/health          # redis up? which modules are loaded?
   curl localhost:8080/docs            # the interactive API browser
   curl localhost:8080/demo/scenario   # exercises all 13 Redis data types
   curl localhost:8080/demo/keys       # what those types wrote

4. Verify the release:

   helm test {{ .Release.Name }} -n {{ .Release.Namespace }}
   helm status {{ .Release.Name }} -n {{ .Release.Namespace }}
